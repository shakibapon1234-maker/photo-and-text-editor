            return;
          }
          if (e.error !== 'aborted') {
            showVoiceToast('Voice error: ' + (e.error || 'unknown'), 'warning', 4000);
          }
        };

        recognition.onend = () => {
          if (isVoiceActive && recognition) {
            try {
              recognition.start();
            } catch (_) {
              setTimeout(() => {
                if (isVoiceActive && recognition) try { recognition.start(); } catch(_) {}
              }, 400);
            }
          } else {
            const btn = document.getElementById('voiceBtn');
            if (btn) btn.classList.remove('voice-on');
          }
        };

        recognition.start();
      } catch (err) {
        console.error(err);
        showVoiceToast('Voice control error: ' + err.message, 'error', 5000);
        stopVoice();
      }
    }

    document.getElementById('voiceBtn').addEventListener('click', () => {
      isVoiceActive ? stopVoice() : startVoice();
    });

        const data = await res.json();
        if (data && data.command && data.command.id && String(data.command.id) !== String(lastExecutedCmdId)) {
          if (lastExecutedCmdId === 0) { lastExecutedCmdId = data.command.id; return; }
          lastExecutedCmdId = data.command.id;
          const cmd = data.command;
          if (cmd.action === 'next') advance(1);
          else if (cmd.action === 'prev') advance(-1);
          else if (cmd.action === 'fullscreen') document.getElementById('fsBtn').click();
